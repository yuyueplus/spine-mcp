cmake_minimum_required(VERSION 3.21)
project(spine_live_bgfx VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Static dependencies are linked into the native shared library on Linux.
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(LIVE_DEPS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.cache" CACHE PATH "Pinned dependency checkouts created by scripts/build.mjs")
foreach(dep bgfx.cmake bgfx bx bimg spine-runtimes json)
  if(NOT EXISTS "${LIVE_DEPS_DIR}/${dep}/.git")
    message(FATAL_ERROR "Missing ${dep}; run node scripts/build.mjs first")
  endif()
endforeach()
set(BGFX_DIR "${LIVE_DEPS_DIR}/bgfx" CACHE PATH "" FORCE)
set(BX_DIR "${LIVE_DEPS_DIR}/bx" CACHE PATH "" FORCE)
set(BIMG_DIR "${LIVE_DEPS_DIR}/bimg" CACHE PATH "" FORCE)
set(BGFX_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(BGFX_BUILD_TOOLS_SHADER ON CACHE BOOL "" FORCE)
set(BGFX_BUILD_TOOLS_GEOMETRY OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_TOOLS_TEXTURE OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_TOOLS_BIN2C OFF CACHE BOOL "" FORCE)
set(BGFX_INSTALL OFF CACHE BOOL "" FORCE)
set(BGFX_CONFIG_RENDERER_WEBGPU OFF CACHE BOOL "" FORCE)
set(BGFX_CONFIG_VIDEO OFF CACHE BOOL "" FORCE)
set(BGFX_WITH_WAYLAND OFF CACHE BOOL "" FORCE)
set(SPINE_SET_COMPILER_FLAGS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LIVE_DEPS_DIR}/bgfx.cmake" "${CMAKE_BINARY_DIR}/bgfx")
add_subdirectory("${LIVE_DEPS_DIR}/spine-runtimes/spine-cpp" "${CMAKE_BINARY_DIR}/spine")

set(shader_outputs)
set(shader_profiles spirv:linux glsl:linux)
if(WIN32)
  list(APPEND shader_profiles dx11:windows)
elseif(APPLE)
  list(APPEND shader_profiles metal:osx)
endif()
foreach(spec ${shader_profiles})
  string(REPLACE ":" ";" pair "${spec}")
  list(GET pair 0 profile_name)
  list(GET pair 1 platform_name)
  if(profile_name STREQUAL "dx11")
    set(profile s_5_0)
  elseif(profile_name STREQUAL "glsl")
    set(profile 430)
  else()
    set(profile "${profile_name}")
  endif()
  foreach(shader_name vs_spine fs_spine fs_nv12)
    string(SUBSTRING "${shader_name}" 0 2 kind)
    if(kind STREQUAL "vs")
      set(shader_type vertex)
    else()
      set(shader_type fragment)
    endif()
    set(output "${CMAKE_BINARY_DIR}/shaders/${profile_name}/${shader_name}.bin")
    add_custom_command(OUTPUT "${output}"
      COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/shaders/${profile_name}"
      COMMAND $<TARGET_FILE:shaderc> -f "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader_name}.sc" -o "${output}"
        --type ${shader_type} --platform ${platform_name} --profile ${profile}
        --varyingdef "${CMAKE_CURRENT_SOURCE_DIR}/shaders/varying.def.sc" -i "${BGFX_DIR}/src"
      DEPENDS shaderc "shaders/${shader_name}.sc" "shaders/varying.def.sc"
      VERBATIM)
    list(APPEND shader_outputs "${output}")
  endforeach()
endforeach()
add_custom_target(live-shaders DEPENDS ${shader_outputs})
add_library(spine-bgfx SHARED src/renderer.cpp)
target_link_libraries(spine-bgfx PRIVATE bgfx bimg_decode spine-cpp)
target_include_directories(spine-bgfx PRIVATE "${LIVE_DEPS_DIR}/json/single_include" "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_definitions(spine-bgfx PRIVATE _CRT_SECURE_NO_WARNINGS SBGFX_LIBRARY_BUILD)
if(MSVC)
  target_compile_options(spine-bgfx PRIVATE /W4 /EHsc /utf-8)
else()
  target_compile_options(spine-bgfx PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(spine-bgfx PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
add_dependencies(spine-bgfx live-shaders)
add_custom_command(TARGET spine-bgfx POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_BINARY_DIR}/shaders" "$<TARGET_FILE_DIR:spine-bgfx>/shaders"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.json" "$<TARGET_FILE_DIR:spine-bgfx>/dependencies.json")
